\begin{frame}{Benchmark generation}
  "Pseudodata": Delphes $\rightarrow$ Delphes \\
  "Fullsim": Delphes $\rightarrow$ CMSSW

  \vspace{0.4cm}
  For both pseudodata and fullsim:
  \begin{enumerate}
    \item Muons: $J/\psi, Z \rightarrow \mu^+\mu^-$
    \item Charged hadrons: 
    \begin{itemize}
      \item Pseudodata: $K_S^0 \rightarrow \pi^+ \pi^-$
      \item Fullsim: $\rho_0 \rightarrow \pi^+ \pi^-$, flattened $\pi^+ \pi^-$
    \end{itemize}
    \item Calorimeters: QCD dijets, unflattened
    \item Electrons: $J/\psi, Z \rightarrow e^+e^-$
  \end{enumerate}
  Then evaluation on held-out $H \rightarrow ZZ^* \rightarrow 4\ell$.

  \vspace{0.4cm}
  On fullsim: $p_T^\text{gen} > 0.25$ Gev, $p_T^\text{reco}$ > 1 GeV

\end{frame}
